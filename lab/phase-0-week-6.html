1Bash: functions, arg parsing, set -euo pipefail, traps

Functions
backup_dir() {
  local src="$1"
  local dest="$2"
  tar -czf "${dest}/backup-$(date +%Y%m%d-%H%M%S).tar.gz" -C "$src" .
  echo "Backed up $src to $dest"
}

backup_dir /etc /srv/backups
Argument parsing with getopts
#!/usr/bin/env bash
while getopts "s:d:v" opt; do
  case "$opt" in
    s) SRC="$OPTARG" ;;
    d) DEST="$OPTARG" ;;
    v) VERBOSE=1 ;;
    *) echo "Usage: $0 -s source -d dest [-v]"; exit 1 ;;
  esac
done

[ -z "$SRC" ] && { echo "Error: -s source is required"; exit 1; }
echo "Backing up $SRC to ${DEST:-/srv/backups}"
set -euo pipefail — fail loudly instead of silently
#!/usr/bin/env bash
set -euo pipefail
# -e: exit immediately if any command fails
# -u: treat unset variables as an error, not an empty string
# -o pipefail: a pipeline fails if ANY command in it fails, not just the last one

# Without pipefail, this would report success even if grep found nothing:
cat missing-file.txt | grep "error"   # now correctly exits non-zero
Traps — guaranteed cleanup
#!/usr/bin/env bash
set -euo pipefail

TMPDIR=$(mktemp -d)
cleanup() {
  echo "Cleaning up $TMPDIR"
  rm -rf "$TMPDIR"
}
trap cleanup EXIT             # runs on normal exit AND on error
trap 'echo "Interrupted"; exit 130' INT

# ... script logic that uses $TMPDIR ...
echo "Working in $TMPDIR"
Checkpoint

Write a script that takes -s and -d flags, uses set -euo pipefail, and cleans up a temp directory via trap even if you kill it mid-run with Ctrl+C.

2Python: file I/O, error handling, venvs, requests

File I/O — always use a context manager
with open("access.log") as f:
    for line in f:
        if "ERROR" in line:
            print(line.strip())

with open("report.txt", "w") as f:
    f.write("Report generated\n")

with open("data.json") as f:
    import json
    data = json.load(f)
Error handling
import sys

try:
    with open("config.yml") as f:
        contents = f.read()
except FileNotFoundError:
    print("config.yml not found", file=sys.stderr)
    sys.exit(1)
except PermissionError as e:
    print(f"Permission denied: {e}", file=sys.stderr)
    sys.exit(1)
else:
    print("Loaded config successfully")
finally:
    print("Done attempting to load config")

requests
import requests

resp = requests.get("https://api.github.com/repos/torvalds/linux", timeout=10)
resp.raise_for_status()              # raises an exception on 4xx/5xx instead of failing silently
data = resp.json()
print(data["stargazers_count"])

resp = requests.post(
    "https://api.example.com/items",
    json={"name": "widget"},
    headers={"Authorization": "Bearer TOKEN"},
    timeout=10,
)
Common trap

Never skip timeout= on a requests call — without it, a hung remote server will hang your script forever. And always call raise_for_status() or check resp.status_code explicitly; a 500 response with a JSON error body will otherwise parse "successfully" and fail somewhere confusing downstream.

3Write 3 automation scripts (log parser, backup script, API poller)

1. Log parser — count HTTP status codes in an access log
#!/usr/bin/env python3
import re
import sys
from collections import Counter

pattern = re.compile(r'"\w+ \S+ HTTP/[\d.]+" (\d{3})')

def main(path):
    counts = Counter()
    with open(path) as f:
        for line in f:
            m = pattern.search(line)
            if m:
                counts[m.group(1)] += 1
    for status, n in sorted(counts.items()):
        print(f"{status}: {n}")

if __name__ == "__main__":
    main(sys.argv[1] if len(sys.argv) > 1 else "/var/log/nginx/access.log")
2. Backup script — tar with timestamp, keep the last 7
#!/usr/bin/env bash
set -euo pipefail

SRC="/etc"
DEST="/srv/backups"
STAMP=$(date +%Y%m%d-%H%M%S)
FILE="${DEST}/etc-${STAMP}.tar.gz"

mkdir -p "$DEST"
tar -czf "$FILE" -C "$(dirname "$SRC")" "$(basename "$SRC")"
echo "Created $FILE"

# Rotate: keep only the 7 most recent backups
ls -1t "${DEST}"/etc-*.tar.gz | tail -n +8 | xargs -r rm --
echo "Rotation complete, $(ls "${DEST}"/etc-*.tar.gz | wc -l) backups retained"
3. API poller — check an endpoint on an interval, log status
#!/usr/bin/env python3
import time
import requests
import logging

logging.basicConfig(
    filename="poller.log",
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(message)s",
)

URL = "https://api.example.com/health"
INTERVAL_SECONDS = 60

def poll_once():
    try:
        resp = requests.get(URL, timeout=10)
        logging.info("status=%s latency_ms=%d", resp.status_code, resp.elapsed.microseconds // 1000)
    except requests.RequestException as e:
        logging.error("poll failed: %s", e)

if __name__ == "__main__":
    while True:
        poll_once()
        time.sleep(INTERVAL_SECONDS)
Checkpoint

All three scripts run without errors against real data, and the poller is wired up as a systemd service (see Week 2) instead of left running in a terminal tab.

First API call
import boto3

s3 = boto3.client("s3")
response = s3.list_buckets()
for bucket in response["Buckets"]:
    print(bucket["Name"])

# Using a resource interface instead of a low-level client:
ec2 = boto3.resource("ec2")
for instance in ec2.instances.all():
    print(instance.id, instance.state["Name"])
Security note

Never commit AWS credentials to Git — not even to a private repo. Use aws configure, environment variables, or (later, in Phase 1) IAM roles. Add .aws/ and any *.pem files to your global .gitignore now, before you have anything to leak.
